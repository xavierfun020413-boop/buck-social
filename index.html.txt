const myId = Math.random().toString(36).substr(2, 9);
const urlParams = new URLSearchParams(window.location.search);
const roomId = urlParams.get('room') || 'drift-global-1';
document.getElementById('room-id').innerText = roomId;

const socket = new WebSocket(`wss://free.piesocket.com/v3/${roomId}?api_key=VCXCEuvhuc6mveSSTfrliLCuFzSca5STAnpZy7uL&notify_self=0`);
const remotePlayers = {};

socket.onmessage = (e) => {
const data = JSON.parse(e.data);
if (data.type === 'update' && data.id !== myId) {
updateRemotePlayer(data);
}
};

function updateRemotePlayer(data) {
if (!remotePlayers[data.id]) {
const mesh = createCarMesh(carData[data.carIdx], true);
scene.add(mesh);
remotePlayers[data.id] = { mesh, lastUpdate: Date.now(), carIdx: data.carIdx };
}

const p = remotePlayers[data.id];
if(p.carIdx !== data.carIdx) {
scene.remove(p.mesh);
p.mesh = createCarMesh(carData[data.carIdx], true);
scene.add(p.mesh);
p.carIdx = data.carIdx;
}

p.mesh.position.set(data.x, data.y, data.z);
p.mesh.rotation.y = data.ry;
p.lastUpdate = Date.now();
}

setInterval(() => {
const now = Date.now();
for (let id in remotePlayers) {
if (now - remotePlayers[id].lastUpdate > 3000) {
scene.remove(remotePlayers[id].mesh);
delete remotePlayers[id];
}
}
document.getElementById('player-count').innerText = Object.keys(remotePlayers).length + 1;
}, 2000);

// --- GAME LOGIC ---
const state = {
coins: 0,
driftPoints: 0,
isDrifting: false,
inShowroom: false,
selectedCarIndex: 0,
previewCarIndex: 0,
ownedCars: [0],
keys: { w: false, a: false, s: false, d: false, space: false },

};

const carData = [

function createCarMesh(data, isGhost = false) {
const group = new THREE.Group();
const bodyMat = new THREE.MeshPhongMaterial({
color: data.color,
transparent: isGhost,
opacity: isGhost ? 0.5 : 1
});

const body = new THREE.Mesh(new THREE.BoxGeometry(1.8 * data.scale, 0.6, 4 * data.scale), bodyMat);
body.position.y = 0.6;
body.castShadow = !isGhost;
group.add(body);

const cabin = new THREE.Mesh(
new THREE.BoxGeometry(1.4 * data.scale, 0.5, 2 * data.scale),
new THREE.MeshPhongMaterial({ color: 0x111111, transparent: isGhost, opacity: isGhost ? 0.3 : 1 })
);
cabin.position.set(0, 1.15, -0.2 * data.scale);
group.add(cabin);

const wheelGeom = new THREE.CylinderGeometry(0.4, 0.4, 0.4, 16);
wheelGeom.rotateZ(Math.PI / 2);
const wheelMat = new THREE.MeshPhongMaterial({ color: 0x111111, transparent: isGhost, opacity: isGhost ? 0.3 : 1 });

group.wheelMeshes = [];
const wheelPos = [{x:-0.9,z:1.2}, {x:0.9,z:1.2}, {x:-0.9,z:-1.2}, {x:0.9,z:-1.2}];
wheelPos.forEach(p => {
const w = new THREE.Mesh(wheelGeom, wheelMat);
w.position.set(p.x * data.scale, 0.4, p.z * data.scale);
group.add(w);
group.wheelMeshes.push(w);
});
return group;
}

let carGroup = createCarMesh(carData[0]);
scene.add(carGroup);

let ground, obstacles = [], ramps = [];
function initMap(mapId) {
if (ground) scene.remove(ground);
obstacles.forEach(o => scene.remove(o));
ramps.forEach(r => scene.remove(r));
obstacles = []; ramps = [];

const config = maps[mapId];
scene.background = new THREE.Color(config.bg);
scene.fog = new THREE.Fog(config.fog, 50, 200);
ground = new THREE.Mesh(new THREE.PlaneGeometry(2000, 2000), new THREE.MeshPhongMaterial({ color: config.ground }));
ground.rotation.x = -Math.PI / 2;
ground.receiveShadow = true;
scene.add(ground);

for(let i=0; i < 30; i++) {
const h = 5 + Math.random() * 20;
const b = new THREE.Mesh(new THREE.BoxGeometry(5, h, 5), new THREE.MeshPhongMaterial({ color: config.obstacles }));
const angle = Math.random() * Math.PI * 2;
const dist = 60 + Math.random() * 250;
b.position.set(Math.cos(angle) * dist, h/2, Math.sin(angle) * dist);
scene.add(b);
obstacles.push(b);
}

const rampGeom = new THREE.BoxGeometry(10, 3, 14);
rampGeom.translate(0, 1.5, 0);
for(let i=0; i < 15; i++) {
const r = new THREE.Mesh(rampGeom, new THREE.MeshPhongMaterial({ color: config.ramp }));
const angle = Math.random() * Math.PI * 2;
const dist = 40 + Math.random() * 200;
r.position.set(Math.cos(angle) * dist, 0, Math.sin(angle) * dist);
r.rotation.y = Math.random() * Math.PI * 2;
scene.add(r);
ramps.push(r);
}
}

function openShowroom() {

state.previewCarIndex = state.selectedCarIndex;
document.getElementById('showroom-ui').style.display = 'flex';
document.getElementById('ui').style.display = 'none';
document.getElementById('menu-buttons').style.display = 'none';

}

function closeShowroom() {
state.inShowroom = false;
document.getElementById('showroom-ui').style.display = 'none';
document.getElementById('ui').style.display = 'block';
document.getElementById('menu-buttons').style.display = 'flex';
if (showroomCar) scene.remove(showroomCar);
scene.remove(carGroup);
carGroup = createCarMesh(carData[state.selectedCarIndex]);
scene.add(carGroup);
}

let showroomCar;
function updateShowroomCar() {
if (showroomCar) scene.remove(showroomCar);
showroomCar = createCarMesh(carData[state.previewCarIndex]);
showroomCar.position.set(carGroup.position.x, 0, carGroup.position.z);
scene.add(showroomCar);
document.getElementById('car-name').innerText = carData[state.previewCarIndex].name;
const btn = document.getElementById('action-btn');
if (state.ownedCars.includes(state.previewCarIndex)) {
btn.innerText = "SELECT"; btn.className = "buy-btn";
} else {
btn.innerText = `BUY: $${carData[state.previewCarIndex].price}`;
btn.className = state.coins >= carData[state.previewCarIndex].price ? "buy-btn" : "buy-btn locked";
}
}

function navCar(dir) {
state.previewCarIndex = (state.previewCarIndex + dir + carData.length) % carData.length;
updateShowroomCar();
}

function handleCarAction() {
if (state.ownedCars.includes(state.previewCarIndex)) {
state.selectedCarIndex = state.previewCarIndex;
closeShowroom();
} else if (state.coins >= carData[state.previewCarIndex].price) {
state.coins -= carData[state.previewCarIndex].price;
state.ownedCars.push(state.previewCarIndex);
state.selectedCarIndex = state.previewCarIndex;
updateShowroomCar();
document.getElementById('coins').innerText = state.coins;
}
}

function toggleMapMenu() {
const order = ['city', 'desert', 'night'];
state.currentMap = order[(order.indexOf(state.currentMap) + 1) % order.length];
document.getElementById('map-btn').innerText = state.currentMap.toUpperCase();
initMap(state.currentMap);
}

let carVelocity = 0, carRotation = 0, driftAngle = 0, carYVelocity = 0;
const friction = 0.99, gravity = 0.018;

function updatePhysics() {
if (state.inShowroom) { if(showroomCar) showroomCar.rotation.y += 0.02; return; }

const currentCar = carData[state.selectedCarIndex];
if (state.keys.w || state.touch.gas) carVelocity += currentCar.accel;
else if (state.keys.s || state.touch.brake) carVelocity -= currentCar.accel * 1.5;

carVelocity = Math.max(-currentCar.maxSpeed * 0.5, Math.min(carVelocity, currentCar.maxSpeed)) * friction;

let steer = 0;
if (state.keys.a || state.touch.left) steer = currentCar.handling;
if (state.keys.d || state.touch.right) steer = -currentCar.handling;

const isHandbraking = state.keys.space;
driftAngle += (steer * Math.abs(carVelocity) * 12 * (isHandbraking ? 3 : 1.2) - driftAngle) * 0.15;
carRotation += steer * (Math.abs(carVelocity) + 0.2) * 0.6;

carYVelocity -= gravity;
carGroup.position.y = Math.max(0, carGroup.position.y + carYVelocity);
if(carGroup.position.y === 0) carYVelocity = 0;

const moveAngle = carRotation + (driftAngle * 0.45);
carGroup.position.x += Math.sin(moveAngle) * carVelocity;
carGroup.position.z += Math.cos(moveAngle) * carVelocity;
carGroup.rotation.y = carRotation + driftAngle;

if (socket.readyState === WebSocket.OPEN) {
socket.send(JSON.stringify({
type: 'update', id: myId, carIdx: state.selectedCarIndex,
x: carGroup.position.x, y: carGroup.position.y, z: carGroup.position.z,
ry: carGroup.rotation.y
}));
}

const driftIntensity = Math.abs(driftAngle) * Math.abs(carVelocity);
if (driftIntensity > 0.05) {
state.isDrifting = true;
state.driftPoints += driftIntensity * 2.5;
document.getElementById('drift-msg').style.opacity = 1;
} else {
if (state.isDrifting) {
state.coins += Math.floor(state.driftPoints);
document.getElementById('coins').innerText = state.coins;
state.driftPoints = 0;
}
state.isDrifting = false;
document.getElementById('drift-msg').style.opacity = 0;
}
document.getElementById('speed').innerText = Math.abs(Math.round(carVelocity * 100));
}

function animate() {
requestAnimationFrame(animate);
updatePhysics();
const target = state.inShowroom ? showroomCar : carGroup;
if(!target) return;
const camOffset = state.inShowroom ?
new THREE.Vector3(0, 3, 8) :
new THREE.Vector3(-Math.sin(carRotation) * 11, 5, -Math.cos(carRotation) * 11);
camera.position.lerp(target.position.clone().add(camOffset), 0.12);
camera.lookAt(target.position);
renderer.render(scene, camera);
}
