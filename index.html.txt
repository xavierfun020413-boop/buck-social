        let auth, db, user, userData;
        const appId = typeof __app_id !== 'undefined' ? __app_id : 'buck-social-app';
        let globalPosts = [];
        let mediaStream = null;
        let mediaRecorder = null;
        let recordedChunks = [];
        let capturedMedia = { url: null, thumb: null, type: null };
        let facingMode = 'user';
        let isRecording = false;
        let captureMode = 'photo'; // 'photo' or 'video'
        let unsubscribeFeed = null;

        const COLLECTIONS = {
            POSTS: `artifacts/${appId}/public/data/posts`,
            USERS: `artifacts/${appId}/public/data/users`
        };

        const initAuth = async () => {
            try {
                let configToUse = myConfig || (typeof __firebase_config !== 'undefined' ? __firebase_config : null);

                if (!configToUse) {
                    document.getElementById('db-status-msg').classList.remove('hidden');
                    setupGuestMode();
                    return;
                }

                const firebaseConfig = typeof configToUse === 'string' ? JSON.parse(configToUse) : configToUse;
                const app = initializeApp(firebaseConfig);
                auth = getAuth(app);
                db = getFirestore(app);

                const performSignIn = async () => {
                    if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
                        return await signInWithCustomToken(auth, __initial_auth_token);
                    } else {
                        return await signInAnonymously(auth);
                    }
                };

                await performSignIn();

                onAuthStateChanged(auth, async (u) => {
                    if (u) {
                        user = u;
                        await syncUser();
                        initFeedListener();
                    } else {
                        setupGuestMode();
                    }
                });
            } catch (error) {
                console.error("Auth Error:", error);
                setupGuestMode();
            }
        };

        function setupGuestMode() {
            user = { uid: 'guest_' + Math.random().toString(36).substr(2, 5) };
            userData = { username: 'Guest', pfp: `https://api.dicebear.com/7.x/avataaars/svg?seed=${user.uid}` };
            updateUI();
            renderFeed();
        }

        async function syncUser() {
            if (!user || !db || user.uid.startsWith('guest_')) return;
            try {
                const userRef = doc(db, COLLECTIONS.USERS, user.uid);
                const snap = await getDoc(userRef);
                if (!snap.exists()) {
                    userData = {
                        uid: user.uid,
                        username: `User_${user.uid.slice(0, 5)}`,
                        pfp: `https://api.dicebear.com/7.x/avataaars/svg?seed=${user.uid}`,
                        createdAt: serverTimestamp()
                    };
                    await setDoc(userRef, userData);
                } else {
                    userData = snap.data();
                }
                updateUI();
            } catch (e) {
                console.error("Sync Error:", e);
            }
        }

        function initFeedListener() {
            if (!user || !db || user.uid.startsWith('guest_')) return;
            if (unsubscribeFeed) unsubscribeFeed();
            const q = query(collection(db, COLLECTIONS.POSTS));
            unsubscribeFeed = onSnapshot(q, (snapshot) => {
                globalPosts = snapshot.docs.map(doc => ({ id: doc.id, ...doc.data() }))
                    .sort((a, b) => (b.createdAt?.seconds || 0) - (a.createdAt?.seconds || 0));
                renderFeed();
                updateUI();
            }, (error) => {
                console.error("Feed Listener Error:", error);
            });
        }

        function updateUI() {
            if (user) {
                const myPosts = globalPosts.filter(p => p.userId === user.uid);
                const totalLikes = myPosts.reduce((sum, p) => sum + (p.likes?.length || 0), 0);
                document.getElementById('stat-bucks-val').innerText = myPosts.length;
                document.getElementById('stat-likes-val').innerText = totalLikes;
                document.getElementById('profile-name').innerText = userData?.username || 'User';
                document.getElementById('profile-pfp').src = userData?.pfp || '';
            }
        }

        window.publishPost = async () => {
            if (!user) return;
            setLoading(true);
            try {
                const caption = document.getElementById('post-title').value;
                const postData = {
                    userId: user.uid,
                    username: userData?.username || 'User',
                    pfp: userData?.pfp || '',
                    type: capturedMedia.type,
                    url: capturedMedia.url,
                    thumb: capturedMedia.thumb,
                    content: caption,
                    likes: [],
                    createdAt: db ? serverTimestamp() : { seconds: Date.now() / 1000 }
                };

                if (db && !user.uid.startsWith('guest_')) {
                    await addDoc(collection(db, COLLECTIONS.POSTS), postData);
                } else {
                    // Fallback for local testing or guest
                    globalPosts.unshift({ id: 'temp_' + Date.now(), ...postData });
                    renderFeed();
                }
                
                document.getElementById('post-title').value = "";
                switchTab('fyp');
                hidePostDetails();
            } catch (e) {
                console.error("Publish Error:", e);
            }
            setLoading(false);
        };

        window.toggleLike = async (postId) => {
            if (!user || !db || user.uid.startsWith('guest_')) return;
            try {
                const postRef = doc(db, COLLECTIONS.POSTS, postId);
                const post = globalPosts.find(p => p.id === postId);
                const isLiked = post.likes.includes(user.uid);
                await updateDoc(postRef, { likes: isLiked ? arrayRemove(user.uid) : arrayUnion(user.uid) });
            } catch (e) {
                console.error("Like Error:", e);
            }
        };

        window.handlePostClick = (element) => {
            const video = element.querySelector('video.feed-video');
            if (!video) return;
            video.paused ? (video.muted = false, video.play()) : video.pause();
        };

        function renderFeed() {
            const container = document.getElementById('tab-fyp');
            if (!globalPosts.length) {
                container.innerHTML = `<div class="flex flex-col items-center justify-center h-full text-zinc-500 p-10 text-center"><p>No posts yet. Tap + to start!</p></div>`;
                return;
            }
            container.innerHTML = globalPosts.map((post) => `
                <div class="snap-item relative h-full w-full overflow-hidden bg-black" onclick="handlePostClick(this)">
                    ${post.type === 'video' ? 
                        `<video src="${post.url}" class="feed-video w-full h-full object-cover" loop playsinline muted></video>` :
                        `<img src="${post.url}" class="w-full h-full object-cover">`
                    }
                    <div class="absolute bottom-24 left-4 right-16">
                        <div class="flex items-center gap-3 mb-3">
                            <img src="${post.pfp}" class="w-10 h-10 rounded-full border border-white/20">
                            <span class="font-bold text-white">@${post.username}</span>
                        </div>
                        <p class="text-sm text-gray-200">${post.content}</p>
                    </div>
                    <div class="absolute right-4 bottom-28 flex flex-col items-center gap-6">
                        <button onclick="event.stopPropagation(); toggleLike('${post.id}')" class="flex flex-col items-center">
                            <i class="fa-solid fa-heart text-3xl ${user && post.likes?.includes(user.uid) ? 'text-red-500' : 'text-white'}"></i>
                            <span class="text-xs font-bold text-white">${post.likes?.length || 0}</span>
                        </button>
                    </div>
                </div>
            `).join('');
        }

        window.switchTab = (tabId) => {
            document.querySelectorAll('section.tab').forEach(s => s.classList.add('hidden'));
            document.getElementById(`tab-${tabId}`).classList.remove('hidden');
            if (tabId === 'camera') startCamera(); else stopCamera();
            if (tabId === 'fyp') renderFeed();
        };

        async function startCamera() {
            try {
                const videoEl = document.getElementById('camera-preview');
                mediaStream = await navigator.mediaDevices.getUserMedia({ 
                    video: { facingMode: facingMode, width: { ideal: 1280 }, height: { ideal: 720 } }, 
                    audio: true 
                });
                videoEl.srcObject = mediaStream;
            } catch (e) {
                console.error("Camera Error:", e);
            }
        }

        function stopCamera() {
            if (mediaStream) mediaStream.getTracks().forEach(t => t.stop());
            isRecording = false;
        }

        window.setCaptureMode = (mode) => {
            captureMode = mode;
            document.getElementById('btn-mode-photo').classList.toggle('text-white', mode === 'photo');
            document.getElementById('btn-mode-photo').classList.toggle('text-zinc-500', mode !== 'photo');
            document.getElementById('btn-mode-video').classList.toggle('text-white', mode === 'video');
            document.getElementById('btn-mode-video').classList.toggle('text-zinc-500', mode !== 'video');
            document.getElementById('record-indicator').classList.toggle('bg-white', mode === 'photo');
            document.getElementById('record-indicator').classList.toggle('bg-red-500', mode === 'video');
        };

        window.handleCaptureAction = () => {
            if (captureMode === 'photo') {
                takePhoto();
            } else {
                if (!isRecording) startRecording();
                else stopRecording();
            }
        };

        function takePhoto() {
            const video = document.getElementById('camera-preview');
            const canvas = document.createElement('canvas');
            canvas.width = video.videoWidth;
            canvas.height = video.videoHeight;
            canvas.getContext('2d').drawImage(video, 0, 0);
            capturedMedia.url = canvas.toDataURL('image/jpeg');
            capturedMedia.thumb = capturedMedia.url;
            capturedMedia.type = 'photo';
            showPostDetails();
        }

        function startRecording() {
            recordedChunks = [];
            mediaRecorder = new MediaRecorder(mediaStream);
            mediaRecorder.ondataavailable = (e) => { if (e.data.size > 0) recordedChunks.push(e.data); };
            mediaRecorder.onstop = () => {
                const blob = new Blob(recordedChunks, { type: 'video/webm' });
                capturedMedia.url = URL.createObjectURL(blob);
                capturedMedia.type = 'video';
                // Create a thumbnail from the first frame
                const video = document.createElement('video');
                video.src = capturedMedia.url;
                video.onloadeddata = () => {
                    const canvas = document.createElement('canvas');
                    canvas.width = video.videoWidth;
                    canvas.height = video.videoHeight;
                    canvas.getContext('2d').drawImage(video, 0, 0);
                    capturedMedia.thumb = canvas.toDataURL('image/jpeg');
                    showPostDetails();
                };
            };
            mediaRecorder.start();
            isRecording = true;
            document.getElementById('record-indicator').classList.add('scale-125', 'animate-pulse');
        }

        function stopRecording() {
            mediaRecorder.stop();
            isRecording = false;
            document.getElementById('record-indicator').classList.remove('scale-125', 'animate-pulse');
        }

        function showPostDetails() {
            document.getElementById('post-details-screen').classList.remove('hidden');
            document.getElementById('post-thumbnail-img').src = capturedMedia.thumb;
            if (capturedMedia.type === 'video') {
                document.getElementById('video-badge').classList.remove('hidden');
            } else {
                document.getElementById('video-badge').classList.add('hidden');
            }
        }

        window.hidePostDetails = () => document.getElementById('post-details-screen').classList.add('hidden');

        function setLoading(l) {
            document.getElementById('loading-overlay').classList.toggle('hidden', !l);
        }

        window.addEventListener('load', initAuth);
